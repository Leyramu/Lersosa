---开始---lersosa---开始---

生成CA私钥：openssl genrsa -aes256 -out ca/lersosa-ca-key.pem 2048

创建自签名的CA证书：openssl req -x509 -new -nodes -key ca/lersosa-ca-key.pem -sha256 -days 3650 -out ca/lersosa-ca-cert.pem -subj "/C=CN/ST=Shandong/L=Jinan/O=Leyramu/OU=Lersosa/CN=Lersosa Root CA"

生成服务器私钥：openssl genrsa -aes256 -out server/lersosa-server-key.pem 2048

生成服务器证书签名请求：openssl req -new -key server/lersosa-server-key.pem -out server/lersosa-server.csr -subj "/C=CN/ST=Shandong/L=Jinan/O=Leyramu/OU=Lersosa/CN=localhost" -addext "subjectAltName = DNS:localhost,IP:127.0.0.1,IP:::1"

签署服务器证书：openssl x509 -req -in server/lersosa-server.csr -CA ca/lersosa-ca-cert.pem -CAkey ca/lersosa-ca-key.pem -CAcreateserial -out server/lersosa-server-cert.pem -days 3650 -sha256

生成客户端私钥：openssl genrsa -aes256 -out client/lersosa-client-key.pem 2048

生成客户端证书签名请求：openssl req -new -key client/lersosa-client-key.pem -out client/lersosa-client.csr -subj "/C=CN/ST=Shandong/L=Jinan/O=Leyramu/OU=Lersosa/CN=localhost" -addext "subjectAltName = DNS:localhost,IP:127.0.0.1,IP:::1"

签署客户端证书：openssl x509 -req -in client/lersosa-client.csr -CA ca/lersosa-ca-cert.pem -CAkey ca/lersosa-ca-key.pem -CAcreateserial -out client/lersosa-client-cert.pem -days 3650 -sha256

---结束---lersosa---结束---



---开始---ca---开始---

生成CA私钥：openssl genrsa -aes256 -out ca/lersosa-ca-key.pem 4096

创建自签名的CA证书：openssl req -x509 -new -nodes -key ca/lersosa-ca-key.pem -sha256 -days 3650 -out ca/lersosa-ca-cert.pem -config ca/lersosa-ca.conf

导入证书新信任库：keytool -importcert -file ca/lersosa-ca-cert.pem -alias "lersosa-ca" -keystore ca/lersosa-ca-truststore.p12 -storetype PKCS12 -storepass "Zcx@223852//" -noprompt

---结束---ca---结束---
